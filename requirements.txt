# Everything a developer needs to work on zicengine from a clone.
#
#     python -m venv .venv
#     .venv\Scripts\activate          (Windows)
#     pip install -r requirements.txt
#
# The two editable installs pull in the engine's own runtime dependencies
# (numpy, OpenCV, Pillow, ezdxf, pyclipper) transitively, so this file only
# has to name the extras: the desktop app, the web backend, and the test
# tooling. See web/backend/requirements.txt and web/frontend/package.json for
# the per-app dependency lists the spec asks for; those are the source of
# truth for deploying one app on its own.

-e ./core
-e ./core[sr]
-e ./desktop

# Web backend.
fastapi>=0.115
uvicorn[standard]>=0.30
python-multipart>=0.0.9

# Test and lint tooling.
pytest>=8.0
pytest-cov>=5.0
ruff>=0.6

# Optional: enables drag-and-drop onto the desktop window. Everything works
# without it -- the app falls back to its Open button.
tkinterdnd2>=0.4 ; sys_platform == "win32"
